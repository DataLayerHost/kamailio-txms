cmake_minimum_required(VERSION 3.16)
project(kamailio_txms VERSION 0.1.0 LANGUAGES C)
include(CTest)
option(TXMS_SANITIZE "Enable address/undefined sanitizers" OFF)
option(TXMS_FUZZ "Build libFuzzer targets with Clang" OFF)
set(LIBTXMS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../libtxms" CACHE PATH "libtxms checkout")
add_subdirectory(${LIBTXMS_SOURCE_DIR} libtxms)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/LIBTXMS_VERSION" TXMS_REQUIRED_LIBRARY_VERSION LIMIT_COUNT 1)
get_directory_property(TXMS_LIBRARY_VERSION DIRECTORY "${LIBTXMS_SOURCE_DIR}" DEFINITION txms_VERSION)
if(NOT TXMS_LIBRARY_VERSION STREQUAL TXMS_REQUIRED_LIBRARY_VERSION)
	message(FATAL_ERROR "Expected libtxms ${TXMS_REQUIRED_LIBRARY_VERSION}; found ${TXMS_LIBRARY_VERSION}")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET libcurl>=7.85 json-c>=0.15 gmime-3.0)
include(CheckCSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES PkgConfig::DEPS)
check_c_source_compiles("#include <gmime/gmime.h>
int main(void) { g_mime_parser_options_set_warning_callback(0,0,0,0); return 0; }" GW_GMIME_WARNING_DESTROY)
unset(CMAKE_REQUIRED_LIBRARIES)
add_library(txms_gateway STATIC src/sms.c src/store.c src/mime.c src/process.c src/rpc.c)
target_include_directories(txms_gateway PUBLIC include)
if(GW_GMIME_WARNING_DESTROY)
	target_compile_definitions(txms_gateway PRIVATE GW_GMIME_WARNING_DESTROY=1)
endif()
target_link_libraries(txms_gateway PUBLIC txms PkgConfig::DEPS)
target_compile_features(txms_gateway PUBLIC c_std_11)
target_compile_options(txms_gateway PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(txms_gateway PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(BUILD_TESTING)
	add_executable(gateway_test tests/test.c)
	target_link_libraries(gateway_test PRIVATE txms_gateway)
	target_compile_options(gateway_test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
	add_test(NAME gateway_unit COMMAND gateway_test)
	add_executable(rpc_test tests/rpc_client.c)
	target_link_libraries(rpc_test PRIVATE txms_gateway)
	find_package(Python3 REQUIRED COMPONENTS Interpreter)
	add_test(NAME rpc_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/rpc_server.py $<TARGET_FILE:rpc_test>)
endif()
if(TXMS_FUZZ)
	target_compile_options(txms_gateway PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
	foreach(kind sms udh mime)
		add_executable(fuzz_${kind} fuzz/${kind}.c)
		target_link_libraries(fuzz_${kind} PRIVATE txms_gateway)
		target_compile_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer,address,undefined)
		target_link_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer,address,undefined)
	endforeach()
endif()
